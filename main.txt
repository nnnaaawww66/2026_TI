import network
import time
import socket
import gc
from media.sensor import *
from media.media import *
from media.display import *

def connect_wifi():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)
    if not wlan.isconnected():
        wlan.connect("NNNAAAWW", "nnaaww20060131")
        for i in range(20):
            if wlan.isconnected():
                return wlan.ifconfig()[0]
            time.sleep(1)
    return wlan.ifconfig()[0] if wlan.isconnected() else None

def send_jpeg_stream():
    ip = connect_wifi()
    if not ip:
        return

    sensor = Sensor()
    sensor.reset()
    sensor.set_framesize(width=640, height=320)
    sensor.set_pixformat(Sensor.RGB888)
    sensor.run()

    display_initialized = False
    try:
        Display.init(Display.ST7701, width=640, height=480, to_ide=True, fps=30)
        display_initialized = True
    except:
        pass

    PC_IP = "10.27.150.166"
    PC_PORT = 8080

    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.settimeout(0.5)

    frame_count = 0
    fps_counter = 0
    fps_timer = time.ticks_ms()
    fps = 0

    CHUNK_SIZE = 1024
    HEADER_MAGIC = b'\xAB\xCD\x12\x34'

    while True:
        try:
            if not network.WLAN(network.STA_IF).isconnected():
                time.sleep(1)
                continue

            img = sensor.snapshot()

            if display_initialized:
                try:
                    Display.show_image(img)
                except:
                    display_initialized = False

            try:
                jpeg_obj = img.compress_for_ide()
            except:
                jpeg_obj = img.compress(quality=101)

            try:
                jpeg_bytes = jpeg_obj.to_bytes()
            except:
                jpeg_bytes = bytes(jpeg_obj)

            jpeg_len = len(jpeg_bytes)

            if jpeg_len > 200 and jpeg_len < 100000:
                header = HEADER_MAGIC + jpeg_len.to_bytes(4, 'big')
                try:
                    sock.sendto(header, (PC_IP, PC_PORT))
                    for offset in range(0, jpeg_len, CHUNK_SIZE):
                        chunk = jpeg_bytes[offset:offset+CHUNK_SIZE]
                        sock.sendto(chunk, (PC_IP, PC_PORT))
                    frame_count += 1
                except:
                    pass

            fps_counter += 1
            if time.ticks_diff(time.ticks_ms(), fps_timer) >= 1000:
                fps = fps_counter
                fps_counter = 0
                fps_timer = time.ticks_ms()
                if frame_count % 60 == 0:
                    gc.collect()

            time.sleep_ms(35)

        except:
            time.sleep(0.5)
            continue

    sensor.stop()
    if display_initialized:
        try:
            Display.deinit()
        except:
            pass
    sock.close()

if __name__ == "__main__":
    send_jpeg_stream()
